Завдання 1. Таймер

<!DOCTYPE html>
<html lang="uk">
<head>
<meta charset="UTF-8">
<title>Завдання</title>
<style>
body{font-family:Arial,sans-serif;padding:12px}
</style>
</head>
<body>
<h2 id="t">0</h2><button id="s">Старт</button> <button id="x">Стоп</button>
<script>
let sec=0,id=null;
s.onclick=()=>{ if(!id) id=setInterval(()=>t.textContent=++sec,1000); };
x.onclick=()=>{ clearInterval(id); id=null; };
</script>
</body>
</html>

Завдання 2. async/await

<!DOCTYPE html>
<html lang="uk">
<head>
<meta charset="UTF-8">
<title>Завдання</title>
<style>
body{font-family:Arial,sans-serif;padding:12px}
</style>
</head>
<body>
<p id="o">Завантаження...</p>
<script>
const load=()=>new Promise(res=>setTimeout(()=>res({user:'Оля',posts:3}),1000));
(async()=>{
  const d=await load();
  document.getElementById('o').textContent=`Користувач ${d.user}, постів: ${d.posts}`;
})();
</script>
</body>
</html>

Завдання 3. localStorage

<!DOCTYPE html>
<html lang="uk">
<head>
<meta charset="UTF-8">
<title>Завдання</title>
<style>
body{font-family:Arial,sans-serif;padding:12px}
</style>
</head>
<body>
<input id="n" placeholder="Ваше ім’я"> <button id="s">Зберегти</button><p id="o"></p>
<script>
const get=()=>{try{return localStorage.getItem('name')}catch(e){return null}};
const show=()=>o.textContent=get()?'Вітаю, '+get()+'!':'Невідомий користувач';
s.onclick=()=>{try{localStorage.setItem('name',n.value)}catch(e){}; show();};
show();
</script>
</body>
</html>
