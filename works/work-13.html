Робота 13. JavaScript: події та валідація

Завдання 1. Лічильник

<!DOCTYPE html>
<html lang="uk">
<head>
<meta charset="UTF-8">
<title>Завдання</title>
<style>
body{font-family:Arial,sans-serif;padding:12px}
</style>
</head>
<body>
<button id="m">−</button> <b id="v">0</b> <button id="p">+</button> <button id="r">Скинути</button>
<script>
let n=0;
const v=document.getElementById('v');
const upd=()=>v.textContent=n;
m.onclick=()=>{n--;upd()};
p.onclick=()=>{n++;upd()};
r.onclick=()=>{n=0;upd()};
</script>
</body>
</html>

Завдання 2. Валідація форми

<!DOCTYPE html>
<html lang="uk">
<head>
<meta charset="UTF-8">
<title>Завдання</title>
<style>
body{font-family:Arial,sans-serif;padding:12px}
</style>
</head>
<body>
<form id="f"><p><input id="e" placeholder="Email"></p><p><input id="p" type="password" placeholder="Пароль"></p><button>Увійти</button></form><p id="msg"></p>
<script>
f.addEventListener('submit',e=>{
  e.preventDefault();
  const okE=/^\S+@\S+\.\S+$/.test(document.getElementById('e').value);
  const okP=document.getElementById('p').value.length>=6;
  msg.textContent=okE&&okP?'Успіх!':'Помилка: '+(!okE?'email ':'')+(!okP?'пароль':'');
  msg.style.color=okE&&okP?'green':'red';
});
</script>
</body>
</html>

Завдання 3. Зміна теми

<!DOCTYPE html>
